28double Atlas::FPS::Now() {
29 return std::chrono::duration<double>(std::chrono::high_resolution_clock::now().time_since_epoch()).count();
30}

32void Atlas::FPS::updateAveragedFPS() {
33 incrementFrameCount();
34
35 // If a second has passed
36 if (hasSecondPassed()) {
37 // Since a second has passed, the FPS is the frame count.
38 // Additionally, the static_cast is used to convert the
39 // frame count to a float for the FPS.
40 setFPS(static_cast<float>(getFrameCount()));
41 // Reset the counters
42 mFrameCount = 0;
43 mPreviousTime = mCurrentTime;
44 }
45}
46
47void Atlas::FPS::updateRealTimeFPS() {
48 setFPS(1.0f / (mCurrentTime - mPreviousTime));
49 setPreviousTime(mCurrentTime);
50 incrementFrameCount();
51}
52
53bool Atlas::FPS::hasSecondPassed() const {
54 return mCurrentTime - mPreviousTime >= 1.0;
55}

57void Atlas::FPS::update() {
58 mCurrentTime = Now();
59
60 if (mFPSCountingMethod == EFPSCountingMethod::Averaged) {
61 updateAveragedFPS();
62 }
63 else {
64 updateRealTimeFPS();
65 }
66}
