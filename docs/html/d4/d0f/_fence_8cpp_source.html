12void Atlas::Fence::ParseAndThrowCreationError(VkResult error) {
13 switch (error)
14 {
15 case VK_ERROR_OUT_OF_DEVICE_MEMORY:
16 throw AException("Failed to create a fence because there was not enough device memory available.");
17 break;
18
19 case VK_ERROR_OUT_OF_HOST_MEMORY:
20 throw AException("Failed to create a fence because there was not enough host memory available.");
21 break;
22
23 // The vkCreateFence() documentation's list of possible error codes states VK_ERROR_VALIDATION_FAILED as a possible error code,
24 // and it is a valid code, but on some situations it is not, therefore, VK_ERROR_VALIDATION_FAILED_EXT is being used instead.
25 // I do not know if this will cause issues down the line.
26 case VK_ERROR_VALIDATION_FAILED_EXT:
27 throw AException("Failed to create a fence because the validation layers detected an error.");
28 break;
29
30 case VK_ERROR_UNKNOWN:
31 throw AException("Failed to create a fence for an unknown reason because VK_ERROR_UNKNOWN was returned by vkCreateFence().");
32 break;
33 default:
34 throw AException("Failed to create a fence for an unknown reason. Please file a bug report.");
35 break;
36 }
37}

86Atlas::Fence::Fence(Device const& ownerDevice, const VkFenceCreateInfo* cCreateInfo, uint64_t timeoutInNS, const VkAllocationCallbacks* cAllocator) : mOwner(ownerDevice), mTimeoutInNS(timeoutInNS), cmAllocationCallbacks(cAllocator) {
87 if (cCreateInfo == nullptr) {
88 throw AException("Cannot create a fence with a null fence create info.");
89 }
90
91 if (!ownerDevice.isValid()) {
92 throw AException("Cannot create a fence with an invalid device. Is it initialized?");
93 }
94

96
97 if (cCreateResult != VK_SUCCESS) {
98 ParseAndThrowCreationError(cCreateResult);
99 }
100 else {
101 InfoLog("Successfully created fence");
102 }

105void Atlas::Fence::destroy(VkDevice ownerDevice) {
106 if (mFence == VK_NULL_HANDLE) {

108 }

124void Atlas::Fence::wait(VkDevice ownerDevice, bool bWaitAll) {
125 if (ownerDevice == VK_NULL_HANDLE) {
126 throw AException("The device passed to use for waiting on the fence is invalid. Did you make sure to pass in the correct device?");
127 }
128
129 const VkResult cWaitResult = vkWaitForFences(ownerDevice, 1, &mFence, bWaitAll, mTimeoutInNS);
130
131 const bool cbHasTimedOut = cWaitResult == VK_TIMEOUT;
132
133 if (cWaitResult != VK_SUCCESS && !cbHasTimedOut) {
134 ParseAndThrowWaitError(cWaitResult);
135 }
136 else if (cbHasTimedOut) {
137 // Despite being treated as a warning, it is still a bug, and should be fixed when thrown because it can
138 // signify some issues in performance and what not.
139 // TODO: Add the mentioned documentation
140 WarnLog("Waiting for fence has timed out. Atlas does not immediately consider this an error, but it is worth checking and attempting to fix. Furthermore, please see the documentation for Fence::wait() in Atlas' docs for more information.");
141 }
142}
