97void Atlas::DescriptorAllocator::init_pool(VkDevice device, uint32_t maxSets, std::span<PoolSizeRatio> poolRatios)
98{
99
100 std::vector<VkDescriptorPoolSize> poolSizes;
101
102 for (PoolSizeRatio ratio : poolRatios) {
103 poolSizes.push_back(VkDescriptorPoolSize{
104 .type = ratio.type,
105 .descriptorCount = uint32_t(ratio.ratio * maxSets)

107 }
108

111 pool_info.maxSets = maxSets;
112 pool_info.poolSizeCount = (uint32_t)poolSizes.size();
113 pool_info.pPoolSizes = poolSizes.data();
114
115 vkCreateDescriptorPool(device, &pool_info, nullptr, &pool);
116}

968 if (const VkResult cWaitResult = vkWaitForFences(mDevice, 1, &currentFrame.renderFence, false, mFencesTimeoutNS); cWaitResult != VK_SUCCESS) {

972 }
973 else if (cWaitResult == VK_ERROR_DEVICE_LOST) {
974 // TODO: Figure out why this happens

976 }
977 else {

979 }

996
997 }

999 }

1001 // reset command buffer
1002 mCurrentDrawData.cmdResetFlags = 0;
1003
1004 //naming it cmd for shorter writing
1005

1069void Atlas::VulkanRenderingBackend::endDrawingMode()
1070{
1071 // Stop drawing if a resize was requested so we can resize the swapchain
1072 if (mbResizeRequested)
1073 {
1074 return;
1075 }
1076
1077 FrameData& currentFrame = getCurrentFrame();
1078
1079 //end render pass
1080 mRenderPassesManager.endDrawingRenderPasses(mCurrentDrawData.cmd, mCurrentDrawData);
1081
1082 //finalize the command buffer (we can no longer add commands, but it can now be executed)
1083 vkEndCommandBuffer(mCurrentDrawData.cmd);
1084
1085 //prepare the submission to the queue.
1086 //we want to wait on the _presentSemaphore, as that semaphore is signaled when the swapchain is ready
1087 //we will signal the _renderSemaphore, to signal that rendering has finished
1088
1089 VkCommandBufferSubmitInfo cmdinfo = SubmitCommandBufferInfo(mCurrentDrawData.cmd);
1090
1091 VkSemaphoreSubmitInfo waitInfo = SubmitSemaphoreInfo(VK_PIPELINE_STAGE_2_COLOR_ATTACHMENT_OUTPUT_BIT_KHR, currentFrame.swapchainSemaphore);
1092 VkSemaphoreSubmitInfo signalInfo = SubmitSemaphoreInfo(VK_PIPELINE_STAGE_2_ALL_GRAPHICS_BIT, currentFrame.renderSemaphore);
1093
1094 VkSubmitInfo2 submit = SubmitInfo(&cmdinfo, &signalInfo, &waitInfo);
1095
1096 //submit command buffer to the queue and execute it.
1097 // _renderFence will now block until the graphic commands finish execution
1098 vkQueueSubmit2(mGraphicsQueue, 1, &submit, currentFrame.renderFence);
1099
1100 // prepare present
1101 // this will put the image we just rendered to into the visible window.
1102 // we want to wait on the _renderSemaphore for that,
1103 // as its necessary that drawing commands have finished before the image is displayed to the user
1104 VkPresentInfoKHR presentInfo = {};
1105 presentInfo.sType = VK_STRUCTURE_TYPE_PRESENT_INFO_KHR;
1106 presentInfo.pNext = nullptr;
1107 presentInfo.pSwapchains = &mSwapchain;
1108 presentInfo.swapchainCount = 1;
1109
1110 presentInfo.pWaitSemaphores = &currentFrame.renderSemaphore;
1111 presentInfo.waitSemaphoreCount = 1;
1112
1113 presentInfo.pImageIndices = &mCurrentDrawData.swapchainImageIndex;
1114
1115 VkResult presentResult = vkQueuePresentKHR(mGraphicsQueue, &presentInfo);
1116 if (presentResult == VK_ERROR_OUT_OF_DATE_KHR) {
1117 mbResizeRequested = true;
1118 }
1119
1120 mFrameTime.update();
1121
1122 mCurrentTime = Now();
1123
1124 //increase the number of frames drawn
1125 mCurrentFrameNumber++;
1126
1127 // reset the current draw data to default values
1128 mCurrentDrawData = {};
1129
1130
1131}

1323
1324 fastgltf::GltfDataBuffer data;
1325 data.loadFromFile(filePath);
1326
1327 constexpr auto gltfOptions = fastgltf::Options::LoadGLBBuffers
1328 | fastgltf::Options::LoadExternalBuffers;
1329
1330 fastgltf::Asset gltf;
1331 fastgltf::Parser parser{};
1332
1333 auto load = parser.loadBinaryGLTF(&data, filePath.parent_path(), gltfOptions);
1334 if (load) {
1335 gltf = std::move(load.get());
1336 }
1337 else {
1338 fmt::print("Failed to load glTF: {} \n", fastgltf::to_underlying(load.error()));
1339 return {};
1340 }
1341
1342 std::vector<std::shared_ptr<MeshAsset>> meshes;
1343
1344 // use the same vectors for all meshes so that the memory doesnt reallocate as
1345 // often
1346 std::vector<uint32_t> indices;
1347 std::vector<Vertex> vertices;
1348 for (fastgltf::Mesh& mesh : gltf.meshes) {
1349 MeshAsset newmesh;
1350
1351 newmesh.name = mesh.name;
1352
1353 // clear the mesh arrays each mesh, we dont want to merge them by error
1354 indices.clear();
1355 vertices.clear();
1356
1357 for (auto&& p : mesh.primitives) {
1358 GeoSurface newSurface;
1359 newSurface.startIndex = (uint32_t)indices.size();
1360 newSurface.count = (uint32_t)gltf.accessors[p.indicesAccessor.value()].count;
1361
1362 size_t initial_vtx = vertices.size();
1363
1364 // load indexes
1365 {
1366 fastgltf::Accessor& indexaccessor = gltf.accessors[p.indicesAccessor.value()];
1367 indices.reserve(indices.size() + indexaccessor.count);
1368
1369 fastgltf::iterateAccessor<std::uint32_t>(gltf, indexaccessor,
1370 [&](std::uint32_t idx) {
1371 indices.push_back(idx + initial_vtx);
1372 });
1373 }
1374
1375 // load vertex positions
1376 {
1377 fastgltf::Accessor& posAccessor = gltf.accessors[p.findAttribute("POSITION")->second];
1378 vertices.resize(vertices.size() + posAccessor.count);
1379
1380 fastgltf::iterateAccessorWithIndex<glm::vec3>(gltf, posAccessor,
1381 [&](glm::vec3 v, size_t index) {
1382 Vertex newvtx;
1383 newvtx.position = v;
1384 newvtx.normal = { 1, 0, 0 };
1385 newvtx.color = glm::vec4{ 1.f };
1386 newvtx.uv_x = 0;
1387 newvtx.uv_y = 0;
1388 vertices[initial_vtx + index] = newvtx;
1389 });
1390 }
1391
1392 // load vertex normals
1393 auto normals = p.findAttribute("NORMAL");
1394 if (normals != p.attributes.end()) {
1395
1396 fastgltf::iterateAccessorWithIndex<glm::vec3>(gltf, gltf.accessors[(*normals).second],
1397 [&](glm::vec3 v, size_t index) {
1398 vertices[initial_vtx + index].normal = v;
1399 });
1400 }
1401
1402 // load UVs
1403 auto uv = p.findAttribute("TEXCOORD_0");
1404 if (uv != p.attributes.end()) {
1405
1406 fastgltf::iterateAccessorWithIndex<glm::vec2>(gltf, gltf.accessors[(*uv).second],
1407 [&](glm::vec2 v, size_t index) {
1408 vertices[initial_vtx + index].uv_x = v.x;
1409 vertices[initial_vtx + index].uv_y = v.y;
1410 });
1411 }
1412
1413 // load vertex colors
1414 auto colors = p.findAttribute("COLOR_0");
1415 if (colors != p.attributes.end()) {
1416
1417 fastgltf::iterateAccessorWithIndex<glm::vec4>(gltf, gltf.accessors[(*colors).second],
1418 [&](glm::vec4 v, size_t index) {
1419 vertices[initial_vtx + index].color = v;
1420 });
1421 }
1422 newmesh.surfaces.push_back(newSurface);
1423 }

1430 }
